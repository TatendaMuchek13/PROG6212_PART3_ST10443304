
# Stage 1: Build the MVC application
FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build

WORKDIR /src

COPY RaceDayWeb/RaceDayWeb.csproj RaceDayWeb/
RUN dotnet restore RaceDayWeb/RaceDayWeb.csproj

COPY RaceDayWeb/ RaceDayWeb/

RUN dotnet publish RaceDayWeb/RaceDayWeb.csproj \
    -c Release \
    -o /app/publish \
    --no-restore

# Stage 2: Run the published application
FROM mcr.microsoft.com/dotnet/aspnet:8.0 AS final

WORKDIR /app

COPY --from=build /app/publish .

ENV ASPNETCORE_URLS=http://+:8080
ENV ASPNETCORE_ENVIRONMENT=Production

EXPOSE 8080

ENTRYPOINT ["dotnet", "RaceDayWeb.dll"]
